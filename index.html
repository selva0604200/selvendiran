# PySpark: Overcoming Data Skew via Key Salting
from pyspark.sql import functions as F

def salted_skew_join(df_large, df_skewed, join_key, salt_factor=8):
    # Add random salt to uniformize executor partitions
    df_skewed_salted = df_skewed.withColumn("_salt", (F.rand() * salt_factor).cast("int"))
    
    # Explode large dimension table with salt array to match
    df_large_exploded = df_large.withColumn("_salt", F.explode(F.array([F.lit(i) for i in range(salt_factor)])))
    
    return df_skewed_salted.join(
        df_large_exploded,
        (df_skewed_salted[join_key] == df_large_exploded[join_key]) &
        (df_skewed_salted["_salt"] == df_large_exploded["_salt"]),
        how="inner"
    ).drop("_salt")
